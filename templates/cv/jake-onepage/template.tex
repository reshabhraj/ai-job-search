% Custom commands
\newcommand{\resumeItem}[1]{
  \item\small{
    {#1 \vspace{-2pt}}
  }
}

% Argument order (the original template's convention; the bold first line is
% what the reader should see first):
%   Experience: \resumeSubheading{Title}{Dates}{Organisation}{Location}
%   Education:  \resumeSubheading{Institution}{Location}{Degree}{Dates}
\newcommand{\resumeSubheading}[4]{
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeSubSubheading}[2]{
    \item
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \textit{\small#1} & \textit{\small #2} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeProjectHeading}[2]{
    \item
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \small#1 & #2 \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeSubItem}[1]{\resumeItem{#1}\vspace{-4pt}}

\renewcommand\labelitemii{$\vcenter{\hbox{\tiny$\bullet$}}$}

\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=0.15in, label={}]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
\newcommand{\resumeItemListStart}{\begin{itemize}}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{-5pt}}

\begin{document}

%----------HEADING----------
% Every contact detail is printed as text: an ATS reads the text layer, not
% link targets, so the LinkedIn and GitHub link text is the URL path itself.
\begin{center}
    \textbf{\Huge \scshape [YOUR_NAME]} \\ \vspace{1pt}
    \small [YOUR_LOCATION] \textbar{} [YOUR_PHONE] \textbar{} \href{mailto:[YOUR_EMAIL]}{\underline{[YOUR_EMAIL]}} \textbar{}
    \href{[YOUR_LINKEDIN_URL]}{\underline{[YOUR_LINKEDIN_PATH]}} \textbar{}
    \href{[YOUR_GITHUB_URL]}{\underline{[YOUR_GITHUB_PATH]}}
\end{center}

%-----------SUMMARY-----------
% Two sentences, at most 45 words: scope and specialisation. Numbers belong in
% the experience bullets.
\section{Summary}
 \begin{itemize}[leftmargin=0.15in, label={}]
    \small{\item{[SUMMARY]}}
 \end{itemize}

%-----------TECHNICAL SKILLS-----------
% Four dense keyword lines; use the posting's own terms where they truthfully apply.
\section{Technical Skills}
 \begin{itemize}[leftmargin=0.15in, label={}]
    \small{\item{
     \textbf{[SKILL_GROUP_1]}{: [SKILLS_1]} \\
     \textbf{[SKILL_GROUP_2]}{: [SKILLS_2]} \\
     \textbf{[SKILL_GROUP_3]}{: [SKILLS_3]} \\
     \textbf{[SKILL_GROUP_4]}{: [SKILLS_4]}
    }}
 \end{itemize}

%-----------EXPERIENCE-----------
% Dates: single ASCII hyphen (Mar 2026 - Present), never -- (the en-dash fails
% the --ascii-dates check).
\section{Experience}
  \resumeSubHeadingListStart

    \resumeSubheading
      {[JOB_TITLE_1]}{[JOB_DATES_1]}
      {[EMPLOYER_1]}{[JOB_LOCATION_1]}
      \resumeItemListStart
        \resumeItem{[BULLET_1_1]}
        \resumeItem{[BULLET_1_2]}
        \resumeItem{[BULLET_1_3]}
      \resumeItemListEnd

    \resumeSubheading
      {[JOB_TITLE_2]}{[JOB_DATES_2]}
      {[EMPLOYER_2]}{[JOB_LOCATION_2]}
      \resumeItemListStart
        \resumeItem{[BULLET_2_1]}
        \resumeItem{[BULLET_2_2]}
      \resumeItemListEnd

  \resumeSubHeadingListEnd

%-----------EDUCATION-----------
\section{Education}
  \resumeSubHeadingListStart
    \resumeSubheading
      {[INSTITUTION_1]}{[EDU_LOCATION_1]}
      {[DEGREE_1]}{[EDU_DATES_1]}
    \resumeSubheading
      {[INSTITUTION_2]}{[EDU_LOCATION_2]}
      {[DEGREE_2]}{[EDU_DATES_2]}
  \resumeSubHeadingListEnd

%-----------LANGUAGES AND HONORS-----------
\section{Languages and Honors}
 \begin{itemize}[leftmargin=0.15in, label={}]
    \small{\item{
     \textbf{Languages}{: [LANGUAGES]} \\
     \textbf{Honors}{: [HONORS]}
    }}
 \end{itemize}

\end{document}
